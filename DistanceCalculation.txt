# filters.py

import math

class LowPassFilter:

    def __init__(self, alpha=0.15):
        self.alpha = alpha
        self.x = 0.0
        self.y = 0.0
        self.z = 0.0
        self.initialized = False

    def update(self, ax, ay, az):

        if not self.initialized:
            self.x = ax
            self.y = ay
            self.z = az
            self.initialized = True

        self.x = self.alpha * ax + (1.0 - self.alpha) * self.x
        self.y = self.alpha * ay + (1.0 - self.alpha) * self.y
        self.z = self.alpha * az + (1.0 - self.alpha) * self.z

        return self.x, self.y, self.z


class HighPassFilter:

    def __init__(self, alpha=0.98):
        self.alpha = alpha
        self.prev_in = [0.0, 0.0, 0.0]
        self.prev_out = [0.0, 0.0, 0.0]

    def update(self, x, y, z):

        ox = self.alpha * (self.prev_out[0] + x - self.prev_in[0])
        oy = self.alpha * (self.prev_out[1] + y - self.prev_in[1])
        oz = self.alpha * (self.prev_out[2] + z - self.prev_in[2])

        self.prev_in = [x, y, z]
        self.prev_out = [ox, oy, oz]

        return ox, oy, oz


class MovingAverage:

    def __init__(self, size=10):

        self.size = size
        self.buffer = []

    def update(self, value):

        self.buffer.append(value)

        if len(self.buffer) > self.size:
            self.buffer.pop(0)

        return sum(self.buffer) / len(self.buffer)


class VectorMovingAverage:

    def __init__(self, size=10):

        self.fx = MovingAverage(size)
        self.fy = MovingAverage(size)
        self.fz = MovingAverage(size)

    def update(self, x, y, z):

        return (
            self.fx.update(x),
            self.fy.update(y),
            self.fz.update(z)
        )


class OneDKalman:

    def __init__(self, q=0.001, r=0.05):

        self.q = q
        self.r = r

        self.x = 0.0
        self.p = 1.0

    def update(self, measurement):

        self.p += self.q

        k = self.p / (self.p + self.r)

        self.x += k * (measurement - self.x)

        self.p *= (1.0 - k)

        return self.x


class VectorKalman:

    def __init__(self):

        self.kx = OneDKalman()
        self.ky = OneDKalman()
        self.kz = OneDKalman()

    def update(self, x, y, z):

        return (
            self.kx.update(x),
            self.ky.update(y),
            self.kz.update(z)
        )


def magnitude(x, y, z):

    return math.sqrt(x * x + y * y + z * z)


def normalize(x, y, z):

    mag = magnitude(x, y, z)

    if mag == 0:
        return 0.0, 0.0, 0.0

    return (
        x / mag,
        y / mag,
        z / mag
    )
